the shorthand rule most people miss

/* Appendix A, on shorthands over coordinated groups:
   "if any longhands have mismatched list lengths … the CSSOM
    representation of the shorthand's value will return the
    empty string." */

el.style.backgroundImage    = "linear-gradient(…)";     // 1 entry
el.style.backgroundPosition = "0px 0px, 10px 10px";     // 2 entries

getComputedStyle(el).background
// Chrome ≤157: "rgba(0, 0, 0, 0) linear-gradient(…) … 0px 0px …"
//              (non-empty — the lists were silently equalised first)
// Chrome 158+: ""  (empty — mismatched lists are not representable)
