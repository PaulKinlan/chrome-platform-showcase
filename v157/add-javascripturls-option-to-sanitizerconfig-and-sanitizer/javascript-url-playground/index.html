the code

// The newly expressible configuration: keep what the unsafe path keeps
// (comments, data-*, <script> elements as inert markup) but still strip
// javascript: navigation URLs.
el.setHTMLUnsafe(untrusted, { sanitizer: { javascriptURLs: false } });

// Safe calls are unaffected: removeUnsafe() forces javascriptURLs to false,
// so this still strips them despite the explicit true:
el.setHTML(untrusted, { sanitizer: { javascriptURLs: true } });
