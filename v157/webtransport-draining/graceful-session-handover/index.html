class DrainingAwareTransportPool {
  constructor(url) {
    this.url = url;
    this.current = this.#connect();
  }

  #connect() {
    const wt = new WebTransport(this.url);
    if ("draining" in wt) {
      wt.draining.then(() => {
        // Server sent HTTP/3 DRAIN_WEBTRANSPORT_SESSION (0x78ae).
        // Immediately warm up a replacement session for new streams:
        const oldSession = wt;
        this.current = this.#connect();
        // Existing streams on oldSession continue until they finish.
      });
    }
    return wt;
  }

  async openBidiStream() {
    await this.current.ready;
    return this.current.createBidirectionalStream();
  }
}
