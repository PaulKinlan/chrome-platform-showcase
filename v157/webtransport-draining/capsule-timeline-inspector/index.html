code path

// Observing all three lifecycle promises on a WebTransport instance:
const wt = new WebTransport("https://edge.example.com/session");

wt.ready.then(() => {
  console.log("[t=0.0s] Extended CONNECT 200 OK; session ready");
});

if ("draining" in wt) {
  wt.draining.then(() => {
    console.log("[0x78ae] DRAIN_WEBTRANSPORT_SESSION received; draining");
  });
}

wt.closed.then(({ closeCode, reason }) => {
  console.log("[0x2843] CLOSE_WEBTRANSPORT_SESSION:", closeCode, reason);
});
