// Sanitizer API — Chrome 146
// setHTML() is the safe replacement for innerHTML = untrustedHTML.
commentPreview.setHTML(userInput);   // default config: no script, no on*, no javascript: URLs

// Narrow it further with a SanitizerConfig (or a Sanitizer built from one).
commentPreview.setHTML(userInput, {
  sanitizer: {
    elements: ['b', 'i', 'code', { name: 'a', attributes: ['href'] }],
    replaceWithChildrenElements: ['p', 'div', 'span'],  // keep their text
  },
});

// setHTML() always applies the safety baseline, even with a permissive config.
// An invalid config (e.g. both elements and removeElements) throws a TypeError.
// setHTMLUnsafe() is the escape hatch that skips the baseline: avoid it for user input.
