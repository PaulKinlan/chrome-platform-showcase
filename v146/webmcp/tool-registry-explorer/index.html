How the API works

// 1. Feature-detect. WebMCP lives on Document, not navigator.
if ('modelContext' in document) {
  const controller = new AbortController();

  // 2. Register a tool. Resolves to undefined; rejects with InvalidStateError
  //    for a duplicate name, an invalid name or an empty description.
  await document.modelContext.registerTool({
    name: 'search_products',           // 1-128 chars of [A-Za-z0-9_.-]
    description: 'Search the product catalogue by keyword.',
    inputSchema: {
      type: 'object',
      properties: {
        query:    { type: 'string', description: 'Search keyword' },
        category: { type: 'string', description: 'Optional category filter' },
        limit:    { type: 'number', description: 'Max results (default 10)' }
      },
      required: ['query']
    },
    // Called when an agent invokes the tool. The browser does NOT validate
    // input against inputSchema, so check it yourself.
    async execute({ query, category, limit = 10 }, { signal }) {
      return { products: await db.search(query, { category, limit, signal }) };
    }
  }, { signal: controller.signal });

  // 3. An in-page agent discovers tools with getTools() and runs one.
  const [tool] = await document.modelContext.getTools();
  const json = await document.modelContext.executeTool(tool, { query: 'lamp' });

  // 4. Aborting the signal unregisters the tool and fires 'toolchange'.
  document.modelContext.addEventListener('toolchange', () => console.log('tools changed'));
  controller.abort();
}
