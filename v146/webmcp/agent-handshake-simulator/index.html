    Protocol sequence

    // Phase 1: Discovery. WebMCP lives on Document.
const tools = await document.modelContext.getTools();
// tools = [Tool { name, title, description, inputSchema, annotations, origin }, ...]

// Phase 2: Selection. The agent's model picks a tool for the task.
const tool = tools.find((t) => t.name === 'search_products');

// Phase 3: Invocation. Pass the Tool object itself, an input object, and
// optionally a signal. The browser does not validate input against inputSchema.
const json = await document.modelContext.executeTool(
  tool,
  { query: 'running shoes', category: 'footwear', limit: 5 },
  { signal: AbortSignal.timeout(10_000) }
);

// Phase 4: Response. json is the JSON-stringified return value of execute().
// A throwing execute() rejects with OperationError; an aborted signal rejects
// with the signal's reason.
const { products } = JSON.parse(json);

// Phase 5: Agent action: show the result, chain another call, or ask the user.
