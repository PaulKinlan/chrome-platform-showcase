Cascade comparison in CSS

/* Base component (Rule 1) */
.styled-btn {
  background: var(--bg-stone);
  color: var(--text-black);
}

/* Theme override (Rule 2) */
.accent-theme .styled-btn {
  background: #1e40af;
  color: #ffffff;
  font-weight: 700;
}

/* revert-rule (Rule 3) — Rolls back background & color to Rule 1,
   preserving Rule 2's font-weight: 700 */
.reset-revert-rule .styled-btn {
  background: revert-rule;  /* → restores Rule 1 stone background */
  color: revert-rule;       /* → restores Rule 1 black text */
  /* font-weight is untouched, so 700 remains active */
}

/* revert — rolls back ALL author rules to user agent defaults */
.reset-revert .styled-btn { all: revert; }

/* initial — resets properties to CSS spec initial values (transparent, normal) */
.reset-initial .styled-btn { background: initial; color: initial; }
