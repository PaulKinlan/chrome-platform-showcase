// TODAY, without buffer_view — one binding per type, or bitcast per element:
@group(0) @binding(0) var<storage> as_u32 : array<u32, 4>;

// WITH buffer_view (requires buffer_view;) — one variable, typed views:
// carve the same 16 bytes into logical u32 / f32 / packed views declared
// once and validated by the compiler — the GPU analogue of:
const bytes = new ArrayBuffer(16);
const u32 = new Uint32Array(bytes);       // no copy
const f32 = new Float32Array(bytes);      // no copy — same storage
