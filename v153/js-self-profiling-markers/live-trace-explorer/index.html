const profiler = new Profiler({ sampleInterval: 10, maxBufferSize: 10000 });
induceWork(600);                 // script burn + layout thrash + canvas paint
const trace = await profiler.stop();

const byMarker = {};
for (const s of trace.samples) {
  const key = "marker" in s ? s.marker : (s.stackId !== undefined ? "(unmarked, stack)" : "(unmarked, no stack)");
  byMarker[key] = (byMarker[key] ?? 0) + 1;
}
// Chrome 153 (cross-origin isolated): script / gc / style / layout / paint / other
// Engines without markers: every key is an "(unmarked …)" bucket — honest.
