why type beats index tables

Without type, “press the touchpad” requires a per-controller table: the DualSense touchpad click surfaces at a vendor-specific index past the standard 17, and other pads differ or lack it entirely. With type, the binding screen just checks captured.type === "trackpad" — one line, every controller, including ones that don't exist yet.

code

function semanticLabel(pad, index) {
  const button = pad.buttons[index];
  if (!('type' in GamepadButton.prototype)) return null; // flag off — fall back to index label
  switch (button.type) {
    case 'trackpad':     return 'Press the touchpad';
    case 'standard':     return STANDARD_NAMES[index]; // safe: standard type ⇒ standard mapping name
    case 'non-standard': return 'Extra button ' + index + ' (controller-specific)';
  }
}
