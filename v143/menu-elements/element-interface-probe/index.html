the exact checks

const item = document.createElement("menuitem");
const supported =
  !(item instanceof HTMLUnknownElement) &&   // parses as its own element
  "disabled" in item &&                      // IDL from HTMLMenuItemElement
  (item.disabled = true, item.matches(":disabled")); // pseudo-class really matches
